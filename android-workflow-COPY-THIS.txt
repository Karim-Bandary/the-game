name: بناء ونشر

# Runs on every push. The deploy job depends on the check job, so a broken
# commit never reaches the phone-testing link — the point of the whole setup.
on:
  push:
    branches: [main]
  workflow_dispatch:

permissions:
  contents: read
  pages: write
  id-token: write

jobs:
  check:
    name: الفحص
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"

      # node is used to syntax-check the inline scripts in the pages
      - uses: actions/setup-node@v4
        with:
          node-version: "20"

      - name: فحص المشروع
        run: python3 tools/check.py

      - name: فحص الفاحص نفسه
        run: python3 tools/test_check.py

  apk:
    name: بناء التطبيق
    needs: check
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: "17"

      - uses: android-actions/setup-android@v3

      - uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: "8.7"

      # Gradle copies the game into assets itself — see android/app/build.gradle.
      - name: بناء نسخة التجربة
        working-directory: android
        run: gradle assembleDebug --no-daemon --stacktrace

      - uses: actions/upload-artifact@v4
        with:
          name: the-game-apk
          path: android/app/build/outputs/apk/debug/app-debug.apk
          if-no-files-found: error

  deploy:
    name: النشر
    needs: check
    runs-on: ubuntu-latest
    # Only the deploy is worth cancelling when a newer push arrives — two
    # deployments cannot run at once. The checks and the app build must finish.
    concurrency:
      group: pages
      cancel-in-progress: true
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - uses: actions/checkout@v4
      - uses: actions/configure-pages@v5
      - uses: actions/upload-pages-artifact@v3
        with:
          path: .
      - id: deployment
        uses: actions/deploy-pages@v4
